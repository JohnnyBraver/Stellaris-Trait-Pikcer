# Grant leader exact XP needed to level up immediately, taking into account current XP (Scope: Leader)
bgltp_grant_level_up_effect = {
	if = {
		limit = { has_base_skill < 10 }
		# Base Game Cygnus 4.5.3 cumulative XP thresholds:
		# Level 1 -> 2: 400 XP
		# Level 2 -> 3: 1100 XP
		# Level 3 -> 4: 2100 XP
		# Level 4 -> 5: 3400 XP
		# Level 5 -> 6: 5000 XP
		# Level 6 -> 7: 6900 XP
		# Level 7 -> 8: 9100 XP
		# Level 8 -> 9: 11600 XP
		# Level 9 -> 10: 14400 XP
		set_variable = { which = bgltp_target_xp value = 400 }
		if = { limit = { has_base_skill = 2 } set_variable = { which = bgltp_target_xp value = 1100 } }
		else_if = { limit = { has_base_skill = 3 } set_variable = { which = bgltp_target_xp value = 2100 } }
		else_if = { limit = { has_base_skill = 4 } set_variable = { which = bgltp_target_xp value = 3400 } }
		else_if = { limit = { has_base_skill = 5 } set_variable = { which = bgltp_target_xp value = 5000 } }
		else_if = { limit = { has_base_skill = 6 } set_variable = { which = bgltp_target_xp value = 6900 } }
		else_if = { limit = { has_base_skill = 7 } set_variable = { which = bgltp_target_xp value = 9100 } }
		else_if = { limit = { has_base_skill = 8 } set_variable = { which = bgltp_target_xp value = 11600 } }
		else_if = { limit = { has_base_skill = 9 } set_variable = { which = bgltp_target_xp value = 14400 } }

		# Read leader's actual current experience
		set_variable = { which = bgltp_cur_xp value = 0 }
		export_trigger_value_to_variable = {
			trigger = has_experience
			variable = bgltp_cur_xp
		}

		# Calculate needed delta: target - current
		subtract_variable = {
			which = bgltp_target_xp
			value = bgltp_cur_xp
		}

		# Add XP if positive, otherwise add 1 XP if already at or beyond threshold
		if = {
			limit = {
				check_variable = {
					which = bgltp_target_xp
					value > 0
				}
			}
			add_experience = bgltp_target_xp
		}
		else = {
			add_experience = 1
		}

		clear_variable = bgltp_target_xp
		clear_variable = bgltp_cur_xp
	}
}

bgltp_grant_xp_to_next_level_minus_one_effect = {
	bgltp_grant_level_up_effect = yes

# Run the full trait scan only when the leader's trait count/points changed since the last scan (Scope: Leader)
# Signature = num_traits * 1000 + num_trait_points (tier upgrades change points, not count).
bgltp_sync_if_changed_effect = {
	set_variable = { which = bgltp_sig_now value = 0 }
	export_trigger_value_to_variable = { trigger = num_traits variable = bgltp_sig_now }
	multiply_variable = { which = bgltp_sig_now value = 1000 }
	set_variable = { which = bgltp_sig_pts value = 0 }
	export_trigger_value_to_variable = { trigger = num_trait_points variable = bgltp_sig_pts }
	change_variable = { which = bgltp_sig_now value = bgltp_sig_pts }
	if = {
		limit = {
			OR = {
				NOT = { has_leader_flag = bgltp_sig_set }
				NOT = { check_variable = { which = bgltp_sig value = bgltp_sig_now } }
			}
		}
		bgltp_sync_leader_traits_effect = yes
		set_variable = { which = bgltp_sig value = bgltp_sig_now }
		set_leader_flag = bgltp_sig_set
	}
	clear_variable = bgltp_sig_now
	clear_variable = bgltp_sig_pts
}
