# Gestalt nodes and gestalt rulers can never leave the council to take field assignments
bgltp_is_gestalt_councilor = {
	OR = {
		is_gestalt_node = yes
		is_gestalt_ruler = yes
	}
}

# Vanilla rule for non-council field traits: excludes Gestalt councilors/restored nodes,
# Imperial/Dictatorial rulers (after game start), and Imperial heirs.
bgltp_can_get_field_traits = {
	can_leader_get_normal_trait_trigger = yes
}

# Largest single XP grant a non-Paragon leader can receive from game scripts is 3000
# (@corporeal_imperial_heir_xp, Shadows of the Shroud), plus headroom for XP gain modifiers.
# Leaders whose XP-to-next-level is within this window are synced every day.
# Note: every cumulative level gap in the base game is <= 2800, so at this value
# every leader below level 10 qualifies. Lower it to trade coverage for speed.
@bgltp_xp_window = 4500

# True if the leader (Scope: Leader) can level up within one worst-case XP grant
bgltp_can_level_up_soon_trigger = {
	OR = {
		AND = { has_base_skill = 1 has_experience >= @[ 400 - bgltp_xp_window ] }
		AND = { has_base_skill = 2 has_experience >= @[ 1100 - bgltp_xp_window ] }
		AND = { has_base_skill = 3 has_experience >= @[ 2100 - bgltp_xp_window ] }
		AND = { has_base_skill = 4 has_experience >= @[ 3400 - bgltp_xp_window ] }
		AND = { has_base_skill = 5 has_experience >= @[ 5000 - bgltp_xp_window ] }
		AND = { has_base_skill = 6 has_experience >= @[ 6900 - bgltp_xp_window ] }
		AND = { has_base_skill = 7 has_experience >= @[ 9100 - bgltp_xp_window ] }
		AND = { has_base_skill = 8 has_experience >= @[ 11600 - bgltp_xp_window ] }
		AND = { has_base_skill = 9 has_experience >= @[ 14400 - bgltp_xp_window ] }
	}
}
