# ==============================================================================
# LEADER LEVEL-UP HANDLER: Native C++ Yellow [+] Picker + Notification Toast
# Fired from on_leader_level_up (Scope = Country, From = Leader)
# ==============================================================================
country_event = {
	id = bgltp.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_ai = no
		has_paragon_dlc = no
		exists = from
		from = {
			is_pool_leader = no
			NOT = {
				has_leader_flag = bgltp_granting_native_pick
				has_leader_flag = retirement_is_default
				has_leader_flag = block_leader_level_up_events
			}
			OR = {
				leader_class = commander
				leader_class = scientist
				leader_class = official
			}
		}
	}
	immediate = {
		from = {
			# 1. Revert the random trait that C++ auto-picked on natural level-up
			bgltp_revert_auto_trait_effect = yes

			# 2. Grant native C++ yellow [+] trait picker by doing set_skill = lvl - 1, add_skill = 1
			set_leader_flag = bgltp_granting_native_pick
			export_trigger_value_to_variable = {
				trigger = has_base_skill
				variable = bgltp_lvl
			}
			change_variable = { which = bgltp_lvl value = -1 }
			set_skill = bgltp_lvl
			add_skill = 1
			clear_variable = bgltp_lvl
			remove_leader_flag = bgltp_granting_native_pick

			# 3. If add_skill = 1 auto-assigned a second trait into traits = "...", revert it!
			# Only revert if an unknown trait was added, leaving available_trait untouched!
			if = {
				limit = { bgltp_has_unknown_trait = yes }
				bgltp_revert_auto_trait_effect = yes
			}
		}
	}
}

# ==============================================================================
# SYNC SCIENTIST ON ANOMALY SUCCESS & PLANET SURVEY (Scope = Ship)
# ==============================================================================
ship_event = {
	id = bgltp.7
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		exists = owner
		owner = {
			is_ai = no
			has_paragon_dlc = no
		}
		OR = {
			exists = leader
			fleet = { exists = leader }
		}
	}
	immediate = {
		if = {
			limit = { exists = leader }
			leader = { bgltp_sync_leader_traits_effect = yes }
		}
		else_if = {
			limit = { fleet = { exists = leader } }
			fleet = { leader = { bgltp_sync_leader_traits_effect = yes } }
		}
	}
}

# ==============================================================================
# DEV TOOL: Console Event event bgltp.99 (Scope = Country)
# Sets owned leaders to (Next Level - 1 XP) so they level up on the next XP tick
# ==============================================================================
country_event = {
	id = bgltp.99
	title = bgltp.99.name
	desc = bgltp.99.desc
	picture = GFX_evt_throne_room
	is_triggered_only = yes

	immediate = {
		# Sync all owned leaders immediately so newly picked traits are registered before testing
		every_owned_leader = {
			bgltp_sync_leader_traits_effect = yes
		}
		random_owned_leader = {
			limit = { is_pool_leader = no is_ruler = yes has_base_skill < 10 }
			save_event_target_as = bgltp_dev_ruler
		}
		random_owned_leader = {
			limit = { is_pool_leader = no is_councilor = yes is_ruler = no has_base_skill < 10 }
			save_event_target_as = bgltp_dev_councilor
		}
		random_owned_leader = {
			limit = { is_pool_leader = no NOT = { has_leader_flag = retirement_is_default } leader_class = commander has_base_skill < 10 }
			save_event_target_as = bgltp_dev_commander
		}
		random_owned_leader = {
			limit = { is_pool_leader = no NOT = { has_leader_flag = retirement_is_default } leader_class = scientist has_base_skill < 10 }
			save_event_target_as = bgltp_dev_scientist
		}
		random_owned_leader = {
			limit = { is_pool_leader = no NOT = { has_leader_flag = retirement_is_default } leader_class = official has_base_skill < 10 }
			save_event_target_as = bgltp_dev_official
		}
	}

	# Option 1: Ruler -> Level Up
	option = {
		name = bgltp.99.ruler
		trigger = { exists = event_target:bgltp_dev_ruler }
		event_target:bgltp_dev_ruler = {
			bgltp_grant_level_up_effect = yes
		}
	}

	# Option 2: Random Commander -> Level Up
	option = {
		name = bgltp.99.commander
		trigger = { exists = event_target:bgltp_dev_commander }
		event_target:bgltp_dev_commander = {
			bgltp_grant_level_up_effect = yes
		}
	}

	# Option 3: Random Scientist -> Level Up
	option = {
		name = bgltp.99.scientist
		trigger = { exists = event_target:bgltp_dev_scientist }
		event_target:bgltp_dev_scientist = {
			bgltp_grant_level_up_effect = yes
		}
	}

	# Option 4: Random Official -> Level Up
	option = {
		name = bgltp.99.official
		trigger = { exists = event_target:bgltp_dev_official }
		event_target:bgltp_dev_official = {
			bgltp_grant_level_up_effect = yes
		}
	}

	# Option 5: ALL Owned Leaders -> Level Up
	option = {
		name = bgltp.99.all
		every_owned_leader = {
			limit = {
				is_pool_leader = no
				NOT = { has_leader_flag = retirement_is_default }
				has_base_skill < 10
				OR = {
					is_ruler = yes
					leader_class = commander
					leader_class = scientist
					leader_class = official
				}
			}
			bgltp_grant_level_up_effect = yes
		}
	}

	# Option 6: Close (Cleanly dismiss dialog)
	option = {
		name = bgltp.99.close
	}
}
