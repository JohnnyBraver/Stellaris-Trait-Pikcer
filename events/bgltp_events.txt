# ==============================================================================
# LEADER LEVEL-UP HANDLER: 4-Trait Weighted Story Event Popup
# Fired from on_leader_level_up (Scope = Country, From = Leader)
# ==============================================================================
country_event = {
	id = bgltp.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_global_flag = game_started
		is_ai = no
		has_paragon_dlc = no
		exists = from
		from = {
			has_base_skill > 1
			has_experience > 0
			is_pool_leader = no
			NOT = {
				has_leader_flag = retirement_is_default
				has_leader_flag = block_leader_level_up_events
			}
			OR = {
				is_ruler = yes
				leader_class = commander
				leader_class = scientist
				leader_class = official
			}
		}
	}
	immediate = {
		from = {
			# 1. Revert the random trait that C++ auto-picked on natural level-up
			bgltp_revert_auto_trait_effect = yes

			# 2. Roll 4 role-weighted eligible traits and open our level-up trait selection popup
			change_variable = { which = bgltp_pending_picks value = 1 }
			if = {
				limit = { NOT = { has_leader_flag = bgltp_selecting_trait } }
				set_leader_flag = bgltp_selecting_trait
				bgltp_roll_4_traits_effect = yes
				leader_event = { id = bgltp.10 }
			}
		}
	}
}

# ==============================================================================
# SYNC SCIENTIST ON ANOMALY SUCCESS & PLANET SURVEY (Scope = Ship)
# ==============================================================================
ship_event = {
	id = bgltp.7
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		exists = owner
		owner = {
			is_ai = no
			has_paragon_dlc = no
		}
		OR = {
			exists = leader
			fleet = { exists = leader }
		}
	}
	immediate = {
		if = {
			limit = { exists = leader }
			leader = { bgltp_sync_leader_traits_effect = yes }
		}
		else_if = {
			limit = { fleet = { exists = leader } }
			fleet = { leader = { bgltp_sync_leader_traits_effect = yes } }
		}
	}
}
